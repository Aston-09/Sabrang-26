rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Check if user is an admin
    function isAdmin() {
      return request.auth != null && 
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    // Check if user is a scanner
    function isScanner() {
      return request.auth != null && 
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'scanner';
    }

    // User Profile: Users can read/write their own profile
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
      allow read: if isAdmin() || isScanner();
    }

    // Events: Public Read, Admin CRUD
    match /events/{eventId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Registrations: Users can CRUD their own, Admin/Scanner can read, Admin/Scanner can update (for attendance)
    match /registrations/{registrationId} {
      allow read, create, delete: if request.auth != null && request.auth.uid == request.resource.data.userId;
      allow read, update: if isAdmin() || isScanner();
    }

    // Coupons: Admin CRUD
    match /coupons/{couponId} {
      allow read, write: if isAdmin();
    }
  }
}
